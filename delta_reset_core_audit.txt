  36:             raise TypeError("Snapshot do Profit deve ser um dict.")
  37: 
  38:         symbol = str(data.get("ativo") or "").strip().upper()
  39:         fingerprint = tuple(self._normalize(data.get(field)) for field in self.FIELDS)
  40:         duplicate = fingerprint == self._last_fingerprint
  41:         symbol_changed = bool(self._last_symbol and symbol and symbol != self._last_symbol)
  42: 
  43:         if not duplicate:
  44:             self._last_fingerprint = fingerprint
  45:         if symbol:
  46:             self._last_symbol = symbol
  47: 
  48:         return SourceIntegrityDecision(
  49:             symbol=symbol,
  50:             duplicate=duplicate,
  51:             symbol_changed=symbol_changed,
  52:             fingerprint=fingerprint,
  53:         )

  56:         self._last_fingerprint = None
  57:         self._last_symbol = ""
  58: 
  59:     @staticmethod
  60:     def _normalize(value):
  61:         if value is None:
  62:             return None
  63:         if isinstance(value, float):
  64:             return round(value, 10)
  65:         if isinstance(value, str):
  66:             return value.strip()
  67:         return value
